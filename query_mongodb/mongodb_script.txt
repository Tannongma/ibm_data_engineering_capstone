-- connect to the mongoDB database through the terminal
```bash
mongosh mongodb://<MONGO_USERNAME>:<MONGO_PASSWORD>@<MONGO_HOST>:<MONGO_PORT>
```
-- check mongo_import and mong_export modules exist, otherwise install
```bash
mongoimport --version
```
OR 
```bash
wget https://fastdl.mongodb.org/tools/db/mongodb-database-tools-ubuntu1804-x86_64-100.3.1.tgz
tar -xf mongodb-database-tools-ubuntu1804-x86_64-100.3.1.tgz
export PATH=$PATH:/path_to/directory/mongodb-database-tools-ubuntu1804-x86_64-100.3.1/bin
echo "done"
```
- download the file to import catalog.json
```bash
wget FILE_URL
```
-- import the file
```bash
mongoimport -u <MONGO_USERNAME> -p <MONGO_PASSWORD> --authenticationDatabase admin --db catalog --collection electronics --file catalog.json --host mongo
``` 

-- list collections in the database catalog (alternatively db.getCollectionInfos() or db.getCollectionNames())
```mongosh
show collections
```
-- create index on field "type"
db.electronics.createIndex( { type: 1 } )

-- find number of smartphones
```mongosh
db.electronics.find( { "type": "laptop" } ).count()
```
-- find the number of smart phones with screen size of 6 inches (alternatively use db.collection.countDocuments())
```mongosh
db.electronics.find( { "type": "smart phone" }, { "screen size": 6 } ).count()
```
-- find out the average screen size of smart phones
```mongosh
db.electronics.aggregate([
   { $match: { "type": "smart phone" } },
   { $group: { _id: "$type", average_screensize: { $avg: "$screen size" } } },
])
```

-- export the fields _id, "type", "model", from the "electronics" collection into a file named electronics.csv

mongoexport -u root -p jxUTDeAi5nuHHJFBTrbirNFl --authenticationDatabase admin \
